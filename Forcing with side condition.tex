\frametitle{$(20/21)$ \hspace{15mm}Generalized case}
\begin{definition}
Let $\alpha < \omega_1$ \textbf{indecomposable}

$\p[\alpha]=\lbrace p=\langle \mathcal{M}_p, f_p, \mathcal{W}_p \rangle \rbrace$ where
\begin{itemize}
\item
$\mathcal{M}_p=\langle M^p_{\xi}\in \el{\aleph_1} \colon \xi \leq \gamma_p < \alpha \rangle$ a \textbf{continuous} $\in$-chain
\item
$f_p: \mathcal{M}_p \longrightarrow H_{\omega_1}$ s.t.

$f_p(M^p_{\xi}) \subset M^p_{\xi+1}$ \textbf{finite}

$f_p(M^p_{\gamma}) \subset H_{\omega_1}$ \textbf{finite}
\item
$\mathcal{W}_p \subset \mathcal{M}_p \text{ s.t. }\forall N \in \mathcal{W}_p,  \res{p}{N}= \langle \mathcal{M}_p \cap N, \res{f_p}{N}, \mathcal{W}_p \cap N \rangle \in N$
\end{itemize}
\end{definition}
\begin{alertblock}{}
$q \leq p$ if and only if
\begin{enumerate}
\item $\mathcal{M}_p \subseteq \mathcal{M}_q$
\item  $f_p(M) \subseteq f_q(M)$ for \textbf{all} $M \in \mathcal{M}_p$
\item  $\mathcal{W}_p \subseteq \mathcal{W}_q$
\end{enumerate}
\pause
\end{alertblock}
\vspace{-2mm}
\begin{block}{}
$C=\{\delta_M \colon \exists p \in G (M \in \mathcal{M}_p)\}$ is a \textbf{club} in $\omega_1$
\end{block}
\end{frame}
\begin{frame}
\frametitle{$(21/21)$ \hspace{20mm} Properness}
\begin{theorem}
$\p[\alpha]$ \textbf{is} $<\alpha$-proper but \textbf{not} $\alpha$-proper
\end{theorem}
\vspace{-2mm}
\pause
\begin{alertblock}{}
$\mathcal{N}=\langle N_{\zeta} \colon \zeta \leq \beta < \alpha \rangle$ and  $p \in \p[\alpha] \cap N_0$ fix

$p'= \langle \mathcal{M}_{p'}, f_{p'}, \mathcal{W}_{p'} \rangle$ such that
\begin{enumerate}
\item $\mathcal{M}_{p'}= \mathcal{M}_p \cup \mathcal{N}$
\item $f_{p'}(M^{p'}_{\xi})=f_p(M^p_{\xi})$ for $\xi \leq \gamma_p$
\item $f_{p'}(N_{\zeta})= \delta_{N_{\zeta}}+1$ for $\zeta \leq \beta$
\item $\mathcal{W}_{p'}= \mathcal{W}_p \cup \{N_{\zeta} \in \mathcal{N} \colon \zeta \text{ is \textbf{not} a \textbf{limit}} \}$
\end{enumerate}
\end{alertblock}
\vspace{-2mm}
\pause
\begin{block}{}
$p'$ is \textbf{strongly} $(N_{\zeta}, \p[\alpha])$-generic if $\zeta$ is \textbf{non-limit}

$p'$ is $(N_{\delta}, \p[\alpha])$-\textbf{generic} if $\delta$ is \textbf{limit}
\end{block}
\vspace{-2mm}
\pause
\begin{alertblock}{}
Let $\mathcal{N}=\langle N_{\xi} \colon \xi \leq \alpha \rangle$ is an $\alpha$-tower and $p \in N_0$

If $ q \leq p$ is $(\mathcal{N}, \p[\alpha])$-generic, then $q \Vdash \dot{C} \text{ is \textbf{not} a \textbf{club}}$
\end{alertblock}
